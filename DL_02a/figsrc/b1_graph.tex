% ---- nút đầu vào ----
\node[inp] (x)  at (0,5.2)   {$x_i$};

\node[op]   (z)  at (3.8,3.6) {$z_i=\theta_0+\theta_1 x_i$};
\node[op]   (yh) at (7.2,3.6) {$\hat{y}_i=\sigma(z_i)$};

% ---- nhãn đạo hàm ----
\node[red!80!black, font=\scriptsize] at (8.9,5.15)
  {$\dfrac{\partial L_i}{\partial \hat{y}_i}=\dfrac{-y_i}{\hat{y}_i}+\dfrac{1-y_i}{1-\hat{y}_i}$};

  {$\dfrac{\partial L_i}{\partial z_i}=\hat{y}_i-y_i$};

  {$\dfrac{\partial L_i}{\partial \theta_1}=(\hat{y}_i-y_i)\,x_i$\\[4pt]

% ---- chú giải ----
