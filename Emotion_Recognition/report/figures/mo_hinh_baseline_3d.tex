% Vẽ một "chồng" bản đồ đặc trưng (feature map)
% #1 x gốc, #2 y gốc, #3 rộng, #4 cao, #5 số tấm, #6 màu
\newcommand{\layer}[6]{%
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\ox}{(#5-\i)*0.085}%  tấm phía sau dịch sang phải
    \pgfmathsetmacro{\oy}{-(#5-\i)*0.045}% và xuống dưới
    \pgfmathsetmacro{\sh}{0.24*(#3)}%      độ nghiêng của tấm
    \fill[#6!18]
      ({#1+\ox},{#2+\oy}) --
      ({#1+\ox+#3},{#2+\oy+\sh}) --
      ({#1+\ox+#3},{#2+\oy+\sh+#4}) --
      ({#1+\ox},{#2+\oy+#4}) -- cycle;
    \draw[#6!65!black, line width=0.5pt]
      ({#1+\ox},{#2+\oy}) --
      ({#1+\ox+#3},{#2+\oy+\sh}) --
      ({#1+\ox+#3},{#2+\oy+\sh+#4}) --
      ({#1+\ox},{#2+\oy+#4}) -- cycle;
  }%
}
